find_package(Doxygen REQUIRED)

set(DOXYGEN_PROJECT_NAME "FyuuAsset")
set(DOXYGEN_PROJECT_BRIEF "FyuuEngine dynamic asset import and persistence library")
set(DOXYGEN_EXTRACT_ALL YES)
set(DOXYGEN_EXTRACT_PRIVATE NO)
set(DOXYGEN_EXTRACT_STATIC NO)
set(DOXYGEN_JAVADOC_AUTOBRIEF YES)
set(DOXYGEN_MARKDOWN_SUPPORT YES)
set(DOXYGEN_FULL_PATH_NAMES NO)
set(DOXYGEN_STRIP_FROM_PATH "${PROJECT_SOURCE_DIR}")
set(DOXYGEN_WARN_IF_UNDOCUMENTED NO)
set(DOXYGEN_GENERATE_HTML YES)
set(DOXYGEN_GENERATE_XML YES)
set(DOXYGEN_GENERATE_LATEX NO)
set(DOXYGEN_RECURSIVE NO)
set(DOXYGEN_FILE_PATTERNS "*.cppm" "*.dox")
set(DOXYGEN_EXTENSION_MAPPING "cppm=C++")

set(FYUU_ASSET_DOCUMENTATION_SOURCES
	${PROJECT_SOURCE_DIR}/module/core.cppm
	${PROJECT_SOURCE_DIR}/module/uuid.cppm
	${PROJECT_SOURCE_DIR}/module/value.cppm
	${PROJECT_SOURCE_DIR}/module/asset.cppm
	${PROJECT_SOURCE_DIR}/module/importer.cppm
	${PROJECT_SOURCE_DIR}/module/factory.cppm
	${PROJECT_SOURCE_DIR}/module/deleter.cppm
)

set(DOXYGEN_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/en")
doxygen_add_docs(
	FyuuAssetDocumentation
	${CMAKE_CURRENT_SOURCE_DIR}/mainpage.dox
	${FYUU_ASSET_DOCUMENTATION_SOURCES}
	COMMENT "Generating FyuuAsset English API documentation"
)

set(DOXYGEN_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/zh-CN")
doxygen_add_docs(
	FyuuAssetDocumentationZhCN
	${CMAKE_CURRENT_SOURCE_DIR}/mainpage.zh-CN.dox
	${FYUU_ASSET_DOCUMENTATION_SOURCES}
	COMMENT "Generating FyuuAsset Simplified Chinese API documentation"
)
